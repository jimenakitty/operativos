#!/bin/bash

# Actualizar el sistema
sudo apt update && sudo apt upgrade -y

# Instalar Docker
if ! command -v docker &> /dev/null; then
  echo "Instalando Docker..."
  sudo apt install -y docker.io
  sudo systemctl enable docker
  sudo systemctl start docker
fi

# Instalar Docker Compose
if ! command -v docker-compose &> /dev/null; then
  echo "Instalando Docker Compose..."
  sudo apt install -y docker-compose
fi

# Crear estructura de proyecto
mkdir -p proyecto/nginx proyecto/web

# Crear index.html
cat <<EOF > proyecto/web/index.html
<!DOCTYPE html>
<html>
<head><title>Servicio Web</title></head>
<body><h1>Hola desde el servidor web</h1></body>
</html>
EOF

# Crear archivo de configuración de NGINX
cat <<EOF > proyecto/nginx/default.conf
upstream backend {
    server web1:80;
    server web2:80;
}

server {
    listen 80;

    location / {
        proxy_pass http://backend;
    }
}
EOF

# Crear archivo docker-compose.yml
cat <<EOF > proyecto/docker-compose.yml
version: '3'
services:
  nginx:
    image: nginx:alpine
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf
    ports:
      - "8080:80"
    depends_on:
      - web1
      - web2

  web1:
    image: nginx:alpine
    volumes:
      - ./web:/usr/share/nginx/html

  web2:
    image: nginx:alpine
    volumes:
      - ./web:/usr/share/nginx/html
EOF

# Ejecutar el entorno
cd proyecto
sudo docker-compose up -d

echo "Arquitectura desplegada en http://localhost:8080"
